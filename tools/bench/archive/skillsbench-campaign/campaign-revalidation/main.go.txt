func main() {
	var repoRoot, checkPath string
	flag.StringVar(&repoRoot, "repo", ".", "repository root")
	flag.StringVar(&checkPath, "check", "", "compare generated bytes with this repository-relative errata path")
	flag.Parse()
	if flag.NArg() != 0 {
		fatal("unexpected positional arguments")
	}
	if _, set := os.LookupEnv("KTX_DB"); set {
		fatal("KTX_DB must be unset")
	}
	result, err := revalidate(repoRoot)
	if err != nil {
		fatal(err.Error())
	}
	data, err := marshalReport(result)
	if err != nil {
		fatal("encode report: " + err.Error())
	}
	if checkPath != "" {
		path, err := safeJoin(repoRoot, checkPath)
		if err != nil {
			fatal("check path: " + err.Error())
		}
		want, err := os.ReadFile(path)
		if err != nil {
			fatal("read checked errata: " + err.Error())
		}
		if !bytes.Equal(want, data) {
			fatal("checked errata is stale")
		}
		fmt.Printf("{\"campaigns\":%d,\"complete_campaigns\":%d,\"valid_pairs\":%d,\"status\":\"ok\"}\n", len(result.Completeness.Campaigns), result.Completeness.CurrentComplete, result.Metrics.ValidPairs)
		return
	}
	if _, err := os.Stdout.Write(data); err != nil {
		fatal("write report: " + err.Error())
	}
}

func marshalReport(result *report) ([]byte, error) {
	data, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		return nil, err
	}
	data = compactShortStringArrays(data, 80)
	return append(data, '\n'), nil
}

func compactShortStringArrays(data []byte, maxLineBytes int) []byte {
	lines := strings.Split(string(data), "\n")
	formatted := make([]string, 0, len(lines))
	for index := 0; index < len(lines); index++ {
		line := lines[index]
		trimmed := strings.TrimSpace(line)
		if !strings.HasSuffix(trimmed, ": [") {
			formatted = append(formatted, line)
			continue
		}

		elements := make([]string, 0)
		closing := ""
		end := index + 1
		for ; end < len(lines); end++ {
			item := strings.TrimSpace(lines[end])
			if item == "]" || item == "]," {
				closing = item
				break
			}
			item = strings.TrimSuffix(item, ",")
			var value string
			if item == "" || json.Unmarshal([]byte(item), &value) != nil {
				break
			}
			elements = append(elements, item)
		}
		if closing == "" || len(elements) == 0 {
			formatted = append(formatted, line)
			continue
		}

		candidate := strings.TrimSuffix(line, "[") + "[" + strings.Join(elements, ", ") + closing
		if len(candidate) > maxLineBytes {
			formatted = append(formatted, line)
			continue
		}
		formatted = append(formatted, candidate)
		index = end
	}
	return []byte(strings.Join(formatted, "\n"))
}

func fatal(message string) {
	fmt.Fprintln(os.Stderr, "campaign revalidation: "+message)
	os.Exit(1)
}

func revalidate(repoRoot string) (*report, error) {
	repoRoot, err := filepath.Abs(repoRoot)
	if err != nil {
		return nil, err
	}
	pins := []authorityRef{
		{Path: archivePath, SHA256: archiveSHA256},
		{Path: artifactPath, SHA256: artifactSHA256},
		{Path: freezeVerifierPath, SHA256: freezeVerifierSHA},
		{Path: historicalPath, SHA256: historicalSHA256},
		{Path: historicalGatePath, SHA256: historicalGateSHA},
		{Path: externalLockPath, SHA256: externalLockSHA},
	}
	for _, pin := range pins {
		if err := verifyPinned(repoRoot, pin); err != nil {
			return nil, err
		}
	}
	rawHashes, err := loadRawManifest(repoRoot)
	if err != nil {
		return nil, err
	}
	if len(rawHashes) != expectedRawFiles {
		return nil, fmt.Errorf("raw manifest count %d, want %d", len(rawHashes), expectedRawFiles)
	}
	for path, want := range rawHashes {
		got, err := hashRegular(repoRoot, path)
		if err != nil || got != want {
			return nil, fmt.Errorf("raw artifact mismatch: %s", path)
		}
	}
	archiveData, err := readPinned(repoRoot, authorityRef{Path: archivePath, SHA256: archiveSHA256})
	if err != nil {
		return nil, err
	}
	var archive archiveDocument
	if err := decodeStrict(archiveData, &archive); err != nil || archive.SchemaVersion != 2 || len(archive.Campaigns) != 10 {
		return nil, errors.New("invalid campaign archive")
	}
	seenCampaigns := map[string]bool{}
	var statuses []campaignStatus
	var pairs []pairMetrics
	removed := 0
	completeAttempts := 0
	for _, campaign := range archive.Campaigns {
		if !canonicalID.MatchString(campaign.ID) || seenCampaigns[campaign.ID] {
			return nil, fmt.Errorf("unsafe or duplicate campaign identity: %s", campaign.ID)
		}
		seenCampaigns[campaign.ID] = true
		status, campaignPairs, err := inspectCampaign(repoRoot, campaign, rawHashes)
		if err != nil {
			return nil, fmt.Errorf("campaign %s: %w", campaign.ID, err)
		}
		statuses = append(statuses, status)
		if status.CurrentComplete {
			completeAttempts += status.Terminal.Valid
			pairs = append(pairs, campaignPairs...)
		}
		if status.HistoricalClaim != nil && *status.HistoricalClaim && !status.CurrentComplete {
			removed++
		}
	}
	sort.Slice(pairs, func(i, j int) bool { return pairs[i].QualifiedPairID < pairs[j].QualifiedPairID })
	completeCount := 0
	for _, status := range statuses {
		if status.CurrentComplete {
			completeCount++
		}
	}
	if completeCount != 1 || len(pairs) != 4 || completeAttempts != 8 || removed != 3 {
		return nil, fmt.Errorf("unexpected corrected totals: complete=%d pairs=%d attempts=%d removed=%d", completeCount, len(pairs), completeAttempts, removed)
	}
	generatorSHA, err := hashRegular(repoRoot, generatorPath)
	if err != nil {
		return nil, err
	}
	metrics := buildCorrectedMetrics(pairs)
	return &report{
		SchemaVersion: "ktx.skillsbench-campaign-errata.v1",
		CorrectedAt:   "2026-08-02",
		Status:        "corrected",
		Supersession: supersession{
			PreferredCorrection: correctionRef{Path: errataPath, ChecksumManifest: "tools/bench/evidence/skillsbench-campaign-errata-evidence/SHA256SUMS"},
			Historical:          []authorityRef{{Path: historicalPath, SHA256: historicalSHA256}, {Path: historicalGatePath, SHA256: historicalGateSHA}},
			Rule:                "Use this errata for current completeness and metrics; retain the historical publications unchanged for provenance.",
		},
		Authority: authority{
			Archive:        authorityRef{Path: archivePath, SHA256: archiveSHA256},
			RawArtifacts:   authorityRef{Path: artifactPath, SHA256: artifactSHA256},
			FreezeVerifier: authorityRef{Path: freezeVerifierPath, SHA256: freezeVerifierSHA},
			ExternalLock:   authorityRef{Path: externalLockPath, SHA256: externalLockSHA},
			Generator:      authorityRef{Path: generatorPath, SHA256: generatorSHA},
			RawFileCount:   len(rawHashes),
		},
		Contract: contract{
			CompleteRequires:     []string{"preflight", "terminal_disposition", "attempt_execution", "result_and_summary", "read_only_database"},
			IdentityFormat:       "campaign_id::attempt_id and campaign_id::pair_id; bare attempt and pair IDs are never global keys",
			FirstArtifactOrder:   []string{"compressed.timestamp ASC", "compressed.id ASC"},

func inspectCampaign(repoRoot string, campaign archiveCampaign, rawHashes map[string]string) (campaignStatus, []pairMetrics, error) {
	status := campaignStatus{CampaignID: campaign.ID, ArchivePath: campaign.Archive, HistoricalClass: campaign.Classification, CorrectedClass: "incomplete", Gaps: []string{}}
	var historical historicalValidation
	if len(campaign.Validation) > 0 && campaign.Validation[0] == '{' && decodeStrict(campaign.Validation, &historical) == nil {
		status.HistoricalClaim = historical.Complete
	}
	if _, err := safeJoin(repoRoot, campaign.Archive); err != nil || !strings.HasPrefix(campaign.Archive, ".bench-data/skillsbench-campaigns/") {
		return status, nil, errors.New("unsafe archive path")
	}
	manifestPath := campaign.Archive + "/manifest.json"
	manifestData, manifestErr := verifiedRaw(repoRoot, manifestPath, rawHashes)
	var manifest Manifest
	manifestOK := manifestErr == nil && decodeStrict(manifestData, &manifest) == nil
	status.SchemaCurrent = manifestOK && manifest.SchemaVersion == currentCampaign && campaign.SchemaVersion == currentCampaign
	if !status.SchemaCurrent {
		status.Gaps = append(status.Gaps, "current_schema")
	}
	manifestHash := digest(manifestData)

	preflightPath := campaign.Archive + "/preflight.json"
	preflightData, preflightErr := verifiedRaw(repoRoot, preflightPath, rawHashes)
	var preflight PreflightEvidence
	preflightOK := manifestOK && preflightErr == nil && decodeStrict(preflightData, &preflight) == nil &&
		preflight.SchemaVersion == manifest.SchemaVersion && preflight.ManifestSHA256 == manifestHash && preflightEligible(preflight.KTXPreflight)
	status.Preflight = layerCount{Expected: 1, Valid: boolInt(preflightOK)}
	if !preflightOK {
		status.Gaps = append(status.Gaps, "preflight")
	}

	schedulePath := campaign.Archive + "/schedule.jsonl"
	scheduleData, scheduleErr := verifiedRaw(repoRoot, schedulePath, rawHashes)
	schedule, scheduleParseErr := decodeJSONLines[ScheduleEntry](scheduleData)
	attemptsPath := campaign.Archive + "/attempts.jsonl"
	attemptsData, attemptsErr := verifiedRaw(repoRoot, attemptsPath, rawHashes)
	attempts, attemptsParseErr := decodeJSONLineRecords[Attempt](attemptsData)
	identityOK := manifestOK && scheduleErr == nil && scheduleParseErr == nil && attemptsErr == nil && attemptsParseErr == nil && len(schedule) > 0 && len(schedule) == len(attempts)
	seenAttempts := map[string]bool{}
	seenPairs := map[string]bool{}
	if identityOK {
		for index, entry := range schedule {
			attempt := attempts[index].Value
			qualified := qualify(campaign.ID, entry.AttemptID)
			if !canonicalID.MatchString(entry.AttemptID) || !canonicalID.MatchString(entry.PairID) || seenAttempts[qualified] ||
				attempt.AttemptID != entry.AttemptID || attempt.PairID != entry.PairID || attempt.Task != entry.Task || attempt.Repetition != entry.Repetition ||
				attempt.Side != entry.Side || attempt.Ordinal != entry.Ordinal || attempt.SourceSHA256 != entry.SourceSHA256 || attempt.OverlaySHA256 != entry.OverlaySHA256 {
				identityOK = false
				break
			}
			seenAttempts[qualified] = true
			seenPairs[qualify(campaign.ID, entry.PairID)] = true
		}
	}
	status.IdentityBound = identityOK
	if !identityOK {
		status.Gaps = append(status.Gaps, "schedule_attempt_identity")
	}
	expected := len(schedule)
	status.Terminal.Expected = expected
	status.Execution.Expected = expected
	status.ResultAndSummary.Expected = expected
	status.Database.Expected = expected

	var inspections []attemptInspection
	if identityOK {
		for index, entry := range schedule {
			inspection := inspectAttempt(repoRoot, campaign, entry, attempts[index], rawHashes)
			inspections = append(inspections, inspection)
			status.Terminal.Valid += boolInt(inspection.terminalOK)
			status.Execution.Valid += boolInt(inspection.executionOK)
			status.ResultAndSummary.Valid += boolInt(inspection.resultOK)
			status.Database.Valid += boolInt(inspection.databaseOK)
			if inspection.terminalOK && inspection.executionOK && inspection.resultOK && inspection.databaseOK {
				status.Bindings = append(status.Bindings, inspection.binding)
			}
		}
	}
	summaryOK := validateCampaignSummary(repoRoot, campaign.Archive, manifestHash, digest(scheduleData), expected, rawHashes)
	if !summaryOK {
		status.ResultAndSummary.Valid = 0
	}
	if status.Terminal.Valid != expected || expected == 0 {
		status.Gaps = append(status.Gaps, "terminal_disposition")
	}
	if status.Execution.Valid != expected || expected == 0 {
		status.Gaps = append(status.Gaps, "attempt_execution")
	}
	if status.ResultAndSummary.Valid != expected || expected == 0 {
		status.Gaps = append(status.Gaps, "result_and_summary")
	}
	if status.Database.Valid != expected || expected == 0 {
		status.Gaps = append(status.Gaps, "database_authority")
	}
	status.CurrentComplete = status.SchemaCurrent && status.IdentityBound && status.Preflight.Valid == 1 && expected > 0 &&
		status.Terminal.Valid == expected && status.Execution.Valid == expected && status.ResultAndSummary.Valid == expected && status.Database.Valid == expected
	if !status.CurrentComplete {
		status.Bindings = nil
		return status, nil, nil
	}
	status.CorrectedClass = "complete"
	pairs, err := pairInspections(campaign.ID, inspections)
	if err != nil {
		return status, nil, err
	}
	if len(pairs)*2 != expected || len(seenPairs) != len(pairs) {
		return status, nil, errors.New("complete campaign has invalid pair topology")
	}
	return status, pairs, nil
}

func inspectAttempt(repoRoot string, campaign archiveCampaign, entry ScheduleEntry, record lineRecord[Attempt], rawHashes map[string]string) attemptInspection {
	attempt := record.Value
	qualifiedAttempt := qualify(campaign.ID, attempt.AttemptID)
	result := attemptInspection{identityOK: true, pairID: attempt.PairID, side: attempt.Side}
	attemptDir := campaign.Archive + "/attempts/" + attempt.AttemptID
	terminalPath := attemptDir + "/terminal.json"
	terminalData, terminalErr := verifiedRaw(repoRoot, terminalPath, rawHashes)
	var terminal AttemptTerminal
	result.terminalOK = terminalErr == nil && decodeStrict(terminalData, &terminal) == nil && terminal.State == "completed" && terminal.Reason == "executor_completed" && terminal.TriggerAttemptID == "" && attempt.Terminal != nil && *attempt.Terminal == terminal

	executionPath := attemptDir + "/execution.json"
	executionData, executionErr := verifiedRaw(repoRoot, executionPath, rawHashes)
	var execution Execution
	result.executionOK = executionErr == nil && decodeStrict(executionData, &execution) == nil && execution.AttemptID == attempt.AttemptID && execution.ExitCode == 0 && !execution.TimedOut && !execution.StdoutTruncated && !execution.StderrTruncated && execution.Version != "" && verifyFileRef(repoRoot, campaign.Archive, execution.Stdout, rawHashes) && verifyFileRef(repoRoot, campaign.Archive, execution.Stderr, rawHashes)

	rolloutPath := campaign.Archive + "/" + attempt.Rollout.Path
	resultPath := rolloutPath + "/result.json"
	benchSummaryPath := attemptDir + "/benchflow/summary.json"
	benchResultsPath := rolloutPath + "/results.jsonl"
	resultData, resultErr := verifiedRaw(repoRoot, resultPath, rawHashes)
	benchSummaryData, benchSummaryErr := verifiedRaw(repoRoot, benchSummaryPath, rawHashes)
	benchResultsData, benchResultsErr := verifiedRaw(repoRoot, benchResultsPath, rawHashes)
	var primary benchResult
	var summary benchSummary
	results, resultsParseErr := decodeJSONLines[benchResultRow](benchResultsData)
	resultRefOK := false
	for _, ref := range attempt.Artifacts {
		if campaign.Archive+"/"+ref.Path == resultPath && verifyFileRef(repoRoot, campaign.Archive, ref, rawHashes) {
			resultRefOK = true
		}
	}
	result.resultOK = resultErr == nil && benchSummaryErr == nil && benchResultsErr == nil && decodeStrict(resultData, &primary) == nil && decodeStrict(benchSummaryData, &summary) == nil && resultsParseErr == nil && len(results) == 1 && resultRefOK && validateAttemptMetrics(attempt) && validateResult(attempt, primary) && validateBenchResultRow(attempt, results[0]) && validateBenchSummary(attempt, summary) && attempt.Imported && attempt.InfrastructureEvaluable && attempt.QualityEvaluable && attempt.Reward != nil && len(attempt.InvalidReasons) == 0

	databasePath := rolloutPath + "/artifacts/ktx.db"
	dbDataHash, dbHashErr := hashRegular(repoRoot, databasePath)
	dbEvidence, dbErr := inspectDatabase(repoRoot, databasePath, qualifiedAttempt)
	result.databaseOK = dbHashErr == nil && dbErr == nil && rawHashes[databasePath] == dbDataHash && validateAdoption(attempt.Adoption, dbEvidence.binding)

	result.binding = attemptBinding{
		QualifiedAttemptID: qualifiedAttempt,
		AttemptRecord:      authorityRef{Path: attemptsRecordPath(campaign.Archive, attempt.AttemptID), SHA256: record.SHA256},
		Execution:          authorityRef{Path: executionPath, SHA256: digest(executionData)},
		Terminal:           authorityRef{Path: terminalPath, SHA256: digest(terminalData)},
		Result:             authorityRef{Path: resultPath, SHA256: digest(resultData)},
		BenchFlowSummary:   authorityRef{Path: benchSummaryPath, SHA256: digest(benchSummaryData)},
		BenchFlowResults:   authorityRef{Path: benchResultsPath, SHA256: digest(benchResultsData)},
		Database:           dbEvidence.binding,
	}
	result.metrics = armMetricsFrom(attempt, dbEvidence.binding, qualifiedAttempt, terminal)
	return result
}

func attemptsRecordPath(campaignPath, attemptID string) string {
	return campaignPath + "/attempts.jsonl#" + attemptID
}

func validateCampaignSummary(repoRoot, campaignPath, manifestHash, scheduleHash string, expected int, rawHashes map[string]string) bool {
	data, err := verifiedRaw(repoRoot, campaignPath+"/summary.json", rawHashes)
	if err != nil {
		return false
	}
	var summary Summary
	if decodeStrict(data, &summary) != nil {
		return false
	}
	return summary.SchemaVersion == currentCampaign && summary.ManifestSHA256 == manifestHash && summary.ScheduleSHA256 == scheduleHash &&
		summary.Scheduled == expected && summary.Imported == expected && summary.QualityEvaluable == expected && summary.Complete && len(summary.Pairs)*2 == expected && len(summary.InvalidAttemptIDs) == 0 && len(summary.UnpairedAttemptIDs) == 0
}

func preflightEligible(values map[string]any) bool {
	if values == nil {
		return false
	}
	eligible, okEligible := values["eligible"].(bool)
	controlled, okControlled := values["controlled_comparison"].(bool)
	direct, okDirect := values["direct_evidence"].(bool)
	created, okCreated := values["artifact_creation"].(bool)
	attempts, okAttempts := numberAsInt(values["model_attempts"])
	return okEligible && eligible && okControlled && controlled && okDirect && direct && okCreated && !created && okAttempts && attempts == 0
}

func validateResult(attempt Attempt, result benchResult) bool {
	reward, rewardOK := result.Rewards["reward"]
	metrics := attempt.Metrics
	return rewardOK && attempt.Reward != nil && sameNumber(reward, *attempt.Reward) && result.TaskName == attempt.Task &&
		samePointer(result.AgentResult.Input, metrics.Input) && samePointer(result.AgentResult.Output, metrics.Output) &&
		samePointer(result.AgentResult.CacheRead, metrics.CacheRead) && samePointer(result.AgentResult.CacheCreation, metrics.CacheCreation) &&
		samePointer(result.AgentResult.Total, metrics.Total) && samePointer(result.AgentResult.CostUSD, metrics.CostUSD) &&
		samePointer(result.TrajectorySummary.Steps, metrics.Turns) && samePointer(result.Timing.Total, metrics.WallSeconds)
}

func validateAttemptMetrics(attempt Attempt) bool {
	metrics := attempt.Metrics
	if attempt.Reward == nil || metrics.Input == nil || metrics.Output == nil || metrics.CacheRead == nil || metrics.CacheCreation == nil ||
		metrics.NewInput == nil || metrics.Total == nil || metrics.Tools == nil || metrics.Turns == nil || metrics.WallSeconds == nil || metrics.CostUSD != nil {
		return false
	}
	return sameNumber(*metrics.NewInput, *metrics.Input+*metrics.CacheCreation) &&
		sameNumber(*metrics.Total, *metrics.Input+*metrics.CacheRead+*metrics.CacheCreation+*metrics.Output)
}

func validateBenchSummary(attempt Attempt, summary benchSummary) bool {

}

	return result.Info.TaskID == attempt.Task && result.Info.TaskName == attempt.Task && result.Info.RolloutName == rolloutName &&
		samePointer(result.Reward, attempt.Reward) && samePointer(result.Metrics.Reward, attempt.Reward) &&
		samePointer(result.Metrics.ToolCalls, attempt.Metrics.Tools) && samePointer(result.TokenUsage.Input, attempt.Metrics.Input) &&
		samePointer(result.TokenUsage.Output, attempt.Metrics.Output) && samePointer(result.TokenUsage.Total, attempt.Metrics.Total) &&
		samePointer(result.Timing.Total, attempt.Metrics.WallSeconds)
}

func validateAdoption(adoption Adoption, binding databaseBinding) bool {

		intPointer(adoption.OriginalBytes, binding.OriginalBytes) && intPointer(adoption.DeliveredBytes, binding.DeliveredBytes) &&
		intPointer(adoption.CompressedArtifactCount, binding.CompressedSourceCount) && intPointer(adoption.DispositionCount, binding.DispositionCount)
}

func inspectDatabase(repoRoot, relativePath, qualifiedAttempt string) (databaseEvidence, error) {
	absolute, err := safeJoin(repoRoot, relativePath)
	if err != nil {
		return databaseEvidence{}, err
	}
	before, err := hashRegular(repoRoot, relativePath)
	if err != nil {
		return databaseEvidence{}, err
	}
	database, err := dbpkg.OpenReadOnly(absolute)
	if err != nil {
		return databaseEvidence{}, err
	}
	defer database.Close()
	var queryOnly, userVersion int
	var integrity string
	if err := database.QueryRow("PRAGMA query_only").Scan(&queryOnly); err != nil || queryOnly != 1 {
		return databaseEvidence{}, errors.New("database is not query-only")
	}
	if err := database.QueryRow("PRAGMA user_version").Scan(&userVersion); err != nil || userVersion != retainedDBSchema {
		return databaseEvidence{}, errors.New("database schema is not retained schema 9")
	}
	if err := database.QueryRow("PRAGMA integrity_check").Scan(&integrity); err != nil || integrity != "ok" {
		return databaseEvidence{}, errors.New("database integrity check failed")
	}
	evidence, err := queryDatabase(database, qualifiedAttempt)
	if err != nil {
		return databaseEvidence{}, err
	}
	if err := database.Close(); err != nil {
		return databaseEvidence{}, err
	}
	after, err := hashRegular(repoRoot, relativePath)
	if err != nil || before != after {
		return databaseEvidence{}, errors.New("database changed during read-only inspection")
	}
	evidence.binding.Path = relativePath
	evidence.binding.SHA256 = after
	evidence.binding.UserVersion = userVersion
	evidence.binding.ReadOnly = true
	evidence.binding.Integrity = integrity
	evidence.binding.SelectedArtifactRule = "compressed.timestamp ASC, compressed.id ASC"
	return evidence, nil
}

func queryDatabase(database *sql.DB, qualifiedAttempt string) (databaseEvidence, error) {
	var evidence databaseEvidence
	knownSessions := map[string]bool{}
	rows, err := database.Query("SELECT id FROM sessions ORDER BY id")
	if err != nil {
		return evidence, err
	}
	for rows.Next() {
		var id string
		if err := rows.Scan(&id); err != nil || !databaseID.MatchString(id) || knownSessions[qualify(qualifiedAttempt, id)] {
			rows.Close()
			return evidence, errors.New("invalid session identity")
		}
		knownSessions[qualify(qualifiedAttempt, id)] = true
		evidence.binding.SessionCount++
	}
	if err := closeRows(rows); err != nil || evidence.binding.SessionCount == 0 {
		return evidence, errors.New("database has no valid session authority")
	}

	type call struct {
		id  string
		raw int64
	}
	calls := map[string]call{}
	rows, err = database.Query("SELECT id, session_id, LENGTH(CAST(output AS BLOB)) FROM tool_calls ORDER BY id")
	if err != nil {
		return evidence, err
	}
	for rows.Next() {
		var id, sessionID string
		var raw sql.NullInt64
		if err := rows.Scan(&id, &sessionID, &raw); err != nil || !databaseID.MatchString(id) || !knownSessions[qualify(qualifiedAttempt, sessionID)] || calls[id].id != "" || !raw.Valid {
			rows.Close()
			return evidence, errors.New("invalid tool-call identity")
		}
		calls[id] = call{id: id, raw: raw.Int64}
		evidence.binding.ToolCallCount++
		evidence.binding.OriginalBytes += raw.Int64
	}
	if err := closeRows(rows); err != nil || evidence.binding.ToolCallCount == 0 {
		return evidence, errors.New("database has no valid tool-call authority")
	}

	type artifact struct {
		id, source, timestamp string
		length                sql.NullInt64
	}
	artifacts := map[string][]artifact{}
	artifactByID := map[string]artifact{}
	rows, err = database.Query("SELECT id, source_id, timestamp, LENGTH(CAST(output AS BLOB)) FROM compressed ORDER BY source_id, timestamp ASC, id ASC")
	if err != nil {
		return evidence, err
	}
	for rows.Next() {
		var item artifact
		if err := rows.Scan(&item.id, &item.source, &item.timestamp, &item.length); err != nil || !databaseID.MatchString(item.id) || calls[item.source].id == "" || artifactByID[item.id].id != "" || item.timestamp == "" {
			rows.Close()
			return evidence, errors.New("invalid compressed-artifact identity")
		}
		artifactByID[item.id] = item
		artifacts[item.source] = append(artifacts[item.source], item)
	}
	if err := closeRows(rows); err != nil {
		return evidence, err
	}
	for source, rows := range artifacts {
		sort.Slice(rows, func(i, j int) bool {
			if rows[i].timestamp == rows[j].timestamp {
				return rows[i].id < rows[j].id
			}
			return rows[i].timestamp < rows[j].timestamp
		})
		artifacts[source] = rows
		evidence.binding.CompressedSourceCount++
	}
	for id, item := range calls {
		delivered := item.raw
		if selected := artifacts[id]; len(selected) > 0 && selected[0].length.Valid {
			delivered = selected[0].length.Int64
		}
		evidence.binding.DeliveredBytes += delivered
	}

	seenDisposition := map[string]bool{}
	rows, err = database.Query("SELECT source_id, disposition, artifact_id FROM compression_dispositions ORDER BY source_id")
	if err != nil {
		return evidence, err
	}
	for rows.Next() {
		var source, disposition string
		var artifactID sql.NullString
		if err := rows.Scan(&source, &disposition, &artifactID); err != nil || calls[source].id == "" || seenDisposition[source] {
			rows.Close()
			return evidence, errors.New("invalid compression disposition identity")
		}
		seenDisposition[source] = true
		evidence.binding.DispositionCount++
		if disposition == "artifact_persisted" {
			artifact := artifactByID[artifactID.String]
			if !artifactID.Valid || artifact.id == "" || artifact.source != source {
				rows.Close()
				return evidence, errors.New("persisted disposition is not bound to its artifact")
			}
		} else if artifactID.Valid || (disposition != "intentional_passthrough" && disposition != "anti_worsening_rejected") {
			rows.Close()
			return evidence, errors.New("invalid compression disposition")
		}
	}
	if err := closeRows(rows); err != nil || evidence.binding.DispositionCount != evidence.binding.ToolCallCount {
		return evidence, errors.New("terminal compression disposition coverage is incomplete")
	}
	return evidence, nil
}

func closeRows(rows *sql.Rows) error {
	if err := rows.Err(); err != nil {
		rows.Close()
		return err
	}
	return rows.Close()
}

func armMetricsFrom(attempt Attempt, database databaseBinding, qualifiedAttempt string, terminal AttemptTerminal) armMetrics {
	metrics := attempt.Metrics
	arm := armMetrics{
		QualifiedAttemptID:  qualifiedAttempt,
		TerminalDisposition: terminal.State + ":" + terminal.Reason,
		Reward:              value(attempt.Reward), OriginalBytes: database.OriginalBytes, DeliveredBytes: database.DeliveredBytes,
		Tokens:       tokenMetrics{Input: value(metrics.Input), NewInput: value(metrics.NewInput), CacheRead: value(metrics.CacheRead), CacheCreation: value(metrics.CacheCreation), Output: value(metrics.Output), Total: value(metrics.Total)},
		RuntimeTools: value(metrics.Tools), KTXCalls: database.ToolCallCount, Turns: value(metrics.Turns), WallSeconds: value(metrics.WallSeconds), CostUSD: metrics.CostUSD,
	}
	if arm.OriginalBytes > 0 {
		arm.CompressionRatio = float64(arm.DeliveredBytes) / float64(arm.OriginalBytes)
		arm.CompressionReduction = float64(arm.OriginalBytes-arm.DeliveredBytes) / float64(arm.OriginalBytes)
	}
	return arm
}

func pairInspections(campaignID string, attempts []attemptInspection) ([]pairMetrics, error) {
	byPair := map[string]map[string]attemptInspection{}
	for _, attempt := range attempts {
		if attempt.side != "baseline" && attempt.side != "candidate" {
			return nil, errors.New("invalid pair side")
		}
		if byPair[attempt.pairID] == nil {
			byPair[attempt.pairID] = map[string]attemptInspection{}
		}
		if _, exists := byPair[attempt.pairID][attempt.side]; exists {
			return nil, errors.New("duplicate pair side")
		}
		byPair[attempt.pairID][attempt.side] = attempt
	}
	ids := make([]string, 0, len(byPair))
	for id := range byPair {
		ids = append(ids, id)
	}
	sort.Strings(ids)
	result := make([]pairMetrics, 0, len(ids))
	for _, id := range ids {
		baseline, baselineOK := byPair[id]["baseline"]
		candidate, candidateOK := byPair[id]["candidate"]
		if !baselineOK || !candidateOK {
			return nil, errors.New("incomplete pair")
		}
		result = append(result, pairMetrics{QualifiedPairID: qualify(campaignID, id), Baseline: baseline.metrics, Candidate: candidate.metrics, CandidateMinusBaseline: subtract(candidate.metrics, baseline.metrics)})
	}
	return result, nil
}

func subtract(candidate, baseline armMetrics) metricDelta {
	return metricDelta{
		Reward:               candidate.Reward - baseline.Reward,
		OriginalBytes:        candidate.OriginalBytes - baseline.OriginalBytes,
		DeliveredBytes:       candidate.DeliveredBytes - baseline.DeliveredBytes,
		CompressionRatio:     candidate.CompressionRatio - baseline.CompressionRatio,
		CompressionReduction: candidate.CompressionReduction - baseline.CompressionReduction,
		Tokens: tokenMetrics{
			Input:         candidate.Tokens.Input - baseline.Tokens.Input,
			NewInput:      candidate.Tokens.NewInput - baseline.Tokens.NewInput,
			CacheRead:     candidate.Tokens.CacheRead - baseline.Tokens.CacheRead,
			CacheCreation: candidate.Tokens.CacheCreation - baseline.Tokens.CacheCreation,
			Output:        candidate.Tokens.Output - baseline.Tokens.Output,
			Total:         candidate.Tokens.Total - baseline.Tokens.Total,
		},
		RuntimeTools: candidate.RuntimeTools - baseline.RuntimeTools,
		KTXCalls:     candidate.KTXCalls - baseline.KTXCalls,
		Turns:        candidate.Turns - baseline.Turns,
		WallSeconds:  candidate.WallSeconds - baseline.WallSeconds,
	}
}

func buildCorrectedMetrics(pairs []pairMetrics) correctedMetrics {
	metrics := correctedMetrics{Scope: "current complete-authority campaigns only", ValidPairs: len(pairs), Pairs: pairs}
	var baselines, candidates []armMetrics
	var deltas []metricDelta
	for _, pair := range pairs {
		baselines = append(baselines, pair.Baseline)
		candidates = append(candidates, pair.Candidate)
		deltas = append(deltas, pair.CandidateMinusBaseline)
		countDirection(pair.CandidateMinusBaseline.Reward, &metrics.DirectionalCounts.Reward)
		countDirection(float64(pair.CandidateMinusBaseline.DeliveredBytes), &metrics.DirectionalCounts.DeliveredBytes)
		countDirection(pair.CandidateMinusBaseline.Tokens.Total, &metrics.DirectionalCounts.TotalTokens)
	}
	metrics.Aggregate = aggregateMetrics{Baseline: aggregateArms(baselines), Candidate: aggregateArms(candidates), CandidateMinusBaseline: deltaAggregate(aggregateDeltas(deltas))}
	metrics.Gate = gateCorrection{State: "incomplete", RequiredValidPairs: 5, ObservedValidPairs: len(pairs), HistoricalDecision: "STOP", CorrectedDecision: nil, Reason: "The terminal-complete evidence set has four valid pairs, not the five required by the predeclared gate.", PaidExpansionAllowed: false}
	return metrics
}

func aggregateArms(values []armMetrics) armAggregate {
	return armAggregate{
		Reward:               dist(valuesOf(values, func(v armMetrics) float64 { return v.Reward })),
		OriginalBytes:        dist(valuesOf(values, func(v armMetrics) float64 { return float64(v.OriginalBytes) })),
		DeliveredBytes:       dist(valuesOf(values, func(v armMetrics) float64 { return float64(v.DeliveredBytes) })),
		CompressionRatio:     dist(valuesOf(values, func(v armMetrics) float64 { return v.CompressionRatio })),
		CompressionReduction: dist(valuesOf(values, func(v armMetrics) float64 { return v.CompressionReduction })),
		TotalTokens:          dist(valuesOf(values, func(v armMetrics) float64 { return v.Tokens.Total })),
		RuntimeTools:         dist(valuesOf(values, func(v armMetrics) float64 { return v.RuntimeTools })),
		KTXCalls:             dist(valuesOf(values, func(v armMetrics) float64 { return float64(v.KTXCalls) })),
		Turns:                dist(valuesOf(values, func(v armMetrics) float64 { return v.Turns })),
		WallSeconds:          dist(valuesOf(values, func(v armMetrics) float64 { return v.WallSeconds })),
	}
}

func aggregateDeltas(values []metricDelta) armAggregate {
	return armAggregate{
		Reward:               dist(deltaValues(values, func(v metricDelta) float64 { return v.Reward })),
		OriginalBytes:        dist(deltaValues(values, func(v metricDelta) float64 { return float64(v.OriginalBytes) })),
		DeliveredBytes:       dist(deltaValues(values, func(v metricDelta) float64 { return float64(v.DeliveredBytes) })),
		CompressionRatio:     dist(deltaValues(values, func(v metricDelta) float64 { return v.CompressionRatio })),
		CompressionReduction: dist(deltaValues(values, func(v metricDelta) float64 { return v.CompressionReduction })),
		TotalTokens:          dist(deltaValues(values, func(v metricDelta) float64 { return v.Tokens.Total })),
		RuntimeTools:         dist(deltaValues(values, func(v metricDelta) float64 { return v.RuntimeTools })),
		KTXCalls:             dist(deltaValues(values, func(v metricDelta) float64 { return float64(v.KTXCalls) })),
		Turns:                dist(deltaValues(values, func(v metricDelta) float64 { return v.Turns })),
		WallSeconds:          dist(deltaValues(values, func(v metricDelta) float64 { return v.WallSeconds })),
	}
}

func valuesOf(values []armMetrics, selectValue func(armMetrics) float64) []float64 {
	result := make([]float64, len(values))
	for i, item := range values {
		result[i] = selectValue(item)
	}
	return result
}

func deltaValues(values []metricDelta, selectValue func(metricDelta) float64) []float64 {
	result := make([]float64, len(values))
	for i, item := range values {
		result[i] = selectValue(item)
	}
	return result
}

func dist(values []float64) distribution {
	sorted := append([]float64(nil), values...)
	sort.Float64s(sorted)
	var sum float64
	for _, item := range sorted {
		sum += item
	}
	median := sorted[len(sorted)/2]
	if len(sorted)%2 == 0 {
		median = (sorted[len(sorted)/2-1] + median) / 2
	}
	return distribution{Mean: sum / float64(len(sorted)), Median: median, Min: sorted[0], Max: sorted[len(sorted)-1]}
}

func countDirection(value float64, counts *lowerEqualHigher) {
	if value < 0 {
		counts.CandidateLower++
	} else if value > 0 {
		counts.CandidateHigher++
	} else {
		counts.Equal++
	}
}

func loadRawManifest(repoRoot string) (map[string]string, error) {
	data, err := readPinned(repoRoot, authorityRef{Path: artifactPath, SHA256: artifactSHA256})
	if err != nil {
		return nil, err
	}
	result := map[string]string{}
	previous := ""
	scanner := bufio.NewScanner(bytes.NewReader(data))
	scanner.Buffer(make([]byte, 64*1024), 4<<20)
	for scanner.Scan() {
		hash, path, ok := strings.Cut(scanner.Text(), "  ")
		if !ok || len(hash) != 64 || path <= previous || result[path] != "" || !strings.HasPrefix(path, ".bench-data/skillsbench-campaigns/") {
			return nil, errors.New("invalid raw artifact manifest")
		}
		if _, err := hex.DecodeString(hash); err != nil {
			return nil, errors.New("invalid raw artifact digest")
		}
		if _, err := safeJoin(repoRoot, path); err != nil {
			return nil, err
		}
		result[path] = hash
		previous = path
	}
	return result, scanner.Err()
}

func verifyPinned(repoRoot string, ref authorityRef) error {
	_, err := readPinned(repoRoot, ref)
	return err
}

func readPinned(repoRoot string, ref authorityRef) ([]byte, error) {
	path, err := safeJoin(repoRoot, ref.Path)
	if err != nil {
		return nil, err
	}
	data, err := readRegular(path, maxJSONBytes)
	if err != nil || digest(data) != ref.SHA256 {
		return nil, fmt.Errorf("pinned authority mismatch: %s", ref.Path)
	}
	return data, nil
}

func verifiedRaw(repoRoot, relative string, rawHashes map[string]string) ([]byte, error) {
	want := rawHashes[relative]
	if want == "" {
		return nil, errors.New("raw file is not frozen")
	}
	path, err := safeJoin(repoRoot, relative)
	if err != nil {
		return nil, err
	}
	data, err := readRegular(path, maxJSONBytes)
	if err != nil || digest(data) != want {
		return nil, errors.New("raw file mismatch")
	}
	return data, nil
}

func verifyFileRef(repoRoot, campaignPath string, ref FileRef, rawHashes map[string]string) bool {
	if ref.Path == "" || ref.SHA256 == "" {
		return false
	}
	data, err := verifiedRaw(repoRoot, campaignPath+"/"+ref.Path, rawHashes)
	return err == nil && digest(data) == ref.SHA256
}

func hashRegular(repoRoot, relative string) (string, error) {
	path, err := safeJoin(repoRoot, relative)
	if err != nil {
		return "", err
	}
	if err := rejectSymlinks(path); err != nil {
		return "", err
	}
	file, err := os.Open(path)
	if err != nil {
		return "", err
	}
	defer file.Close()
	info, err := file.Stat()
	if err != nil || !info.Mode().IsRegular() {
		return "", errors.New("not a regular file")
	}
	h := sha256.New()
	if _, err := io.Copy(h, file); err != nil {
		return "", err
	}
	return hex.EncodeToString(h.Sum(nil)), nil
}

func readRegular(path string, limit int64) ([]byte, error) {
	if err := rejectSymlinks(path); err != nil {
		return nil, err
	}
	info, err := os.Lstat(path)
	if err != nil || !info.Mode().IsRegular() {
		return nil, errors.New("not a non-symlink regular file")
	}
	file, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer file.Close()
	data, err := io.ReadAll(io.LimitReader(file, limit+1))
	if err != nil || int64(len(data)) > limit {
		return nil, errors.New("file exceeds read bound")
	}
	return data, nil
}

func safeJoin(root, relative string) (string, error) {
	clean := filepath.Clean(filepath.FromSlash(relative))
	if clean == "." || filepath.IsAbs(clean) || clean == ".." || strings.HasPrefix(clean, ".."+string(filepath.Separator)) {
		return "", errors.New("unsafe relative path")
	}
	joined := filepath.Join(root, clean)
	rel, err := filepath.Rel(root, joined)
	if err != nil || rel == ".." || strings.HasPrefix(rel, ".."+string(filepath.Separator)) {
		return "", errors.New("path escapes repository")
	}
	return joined, nil
}

func rejectSymlinks(path string) error {
	absolute, err := filepath.Abs(path)
	if err != nil {
		return err
	}
	current := string(filepath.Separator)
	for _, part := range strings.Split(strings.Trim(absolute, string(filepath.Separator)), string(filepath.Separator)) {
		current = filepath.Join(current, part)
		info, err := os.Lstat(current)
		if err != nil {
			return err
		}
		if info.Mode()&os.ModeSymlink != 0 {
			return errors.New("symlink is not allowed")
		}
	}
	return nil
}

func decodeStrict(data []byte, target any) error {
	canonical, ok := canon.Canonicalize(data)
	if !ok {
		return errors.New("invalid or duplicate JSON")
	}
	decoder := json.NewDecoder(bytes.NewReader(canonical))
	decoder.UseNumber()
	return decoder.Decode(target)
}

func decodeJSONLines[T any](data []byte) ([]T, error) {
	records, err := decodeJSONLineRecords[T](data)
	if err != nil {
		return nil, err
	}
	result := make([]T, len(records))
	for i := range records {
		result[i] = records[i].Value
	}
	return result, nil
}

func decodeJSONLineRecords[T any](data []byte) ([]lineRecord[T], error) {
	var records []lineRecord[T]
	scanner := bufio.NewScanner(bytes.NewReader(data))
	scanner.Buffer(make([]byte, 64*1024), maxJSONBytes)
	for scanner.Scan() {
		line := bytes.TrimSpace(scanner.Bytes())
		if len(line) == 0 {
			continue
		}
		var value T
		if err := decodeStrict(line, &value); err != nil {
			return nil, err
		}
		records = append(records, lineRecord[T]{Value: value, SHA256: digest(line)})
	}
	return records, scanner.Err()
}

func digest(data []byte) string {
	sum := sha256.Sum256(data)
	return hex.EncodeToString(sum[:])
}

func qualify(scope, id string) string { return scope + "::" + id }
func boolInt(value bool) int {
	if value {
		return 1
	}
	return 0
}

func value(number *float64) float64 {
	if number == nil {
		return math.NaN()
	}
	return *number
}
func sameNumber(left, right float64) bool { return math.Abs(left-right) < 1e-9 }
func samePointer(left, right *float64) bool {
	return left != nil && right != nil && sameNumber(*left, *right) || left == nil && right == nil
}
func intPointer(value *int64, expected int64) bool { return value != nil && *value == expected }
func numberAsInt(value any) (int, bool) {
	switch number := value.(type) {
	case json.Number:
		parsed, err := number.Int64()
		return int(parsed), err == nil
	case float64:
		return int(number), number == math.Trunc(number)
	default:
		return 0, false
	}
}
